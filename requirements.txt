# ── Install order ────────────────────────────────────────────────────
# Install PyTorch FIRST, with pinned versions. Before this file.
#
#   NVIDIA GPU (Windows/Linux):
#     pip install torch==2.8.0 torchaudio==2.8.0 torchvision==0.23.0 ^
#         --index-url https://download.pytorch.org/whl/cu128
#
#   CPU / macOS:
#     pip install torch==2.8.0 torchaudio==2.8.0 torchvision==0.23.0
#
# Why torch 2.8.0 and not the latest (2.13):
#   whisperx 3.8.6 requires torch~=2.8.0. If you install the newest torch first,
#   installing whisperx below downgrades it to 2.8.0 — and it pulls that from
#   the default PyPI index, which on Windows means the CPU-only wheel (230MB).
#   Your GPU then silently stops being used. So we fetch 2.8.0 from the CUDA
#   index from the start.
#
# Use Python 3.13. whisperx does not support 3.14 yet.
# ─────────────────────────────────────────────────────────────────────

whisperx==3.8.6
pyannote.audio>=4.0.7

# torchcodec is ABI-locked to torch. torch 2.8 wants 0.6 or 0.7.
# whisperx requires <0.8 and pyannote requires >=0.7, so 0.7.x is the only overlap.
torchcodec>=0.7.0,<0.8.0

fastapi>=0.115.0
uvicorn[standard]>=0.30.0
python-multipart>=0.0.9
jinja2>=3.1.0
